From SubworldLibrary's Subworld.cs (namespace SubworldLibrary):

public abstract class Subworld : ModType, ICopyWorldData, ILocalizedModType
{
    public abstract int Width { get; }
    public abstract int Height { get; }
    public abstract List<GenPass> Tasks { get; }

    public virtual WorldGenConfiguration Config => null;
    public virtual int ReturnDestination => -1;      // -1 = main world
    public virtual bool ShouldSave => false;          // default already false
    public virtual bool NoPlayerSaving => false;      // default already false -- DO NOT override to true
    public virtual bool NormalUpdates => false;
    public virtual bool ManualAudioUpdates => false;

    public virtual void OnEnter() { }
    public virtual void OnExit() { }
    public virtual void Update() { }
    public virtual void OnLoad() { }
    public virtual void OnUnload() { }
}

SubworldSystem.LoadSubworld() confirmed call sequence (source-read, 01-RESEARCH.md "Pattern 2"):

Main.maxTilesX = current.Width;
