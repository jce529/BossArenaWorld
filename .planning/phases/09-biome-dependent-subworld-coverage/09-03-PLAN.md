    ```csharp
    using Terraria;
    using Terraria.ID;
    using Terraria.IO;
    using Terraria.WorldBuilding;

    namespace BossArenaSubWorld.Subworlds
    {
    	// Source: decompiled Terraria.Player.UpdateBiomes() / Terraria.SceneMetrics --
    	// ZoneDesert = SceneMetrics.SandTileCount >= SceneMetrics.DesertTileThreshold (1500), a per-tick
    	// weighted sum over TileID.Sets.SandBiome (weight 1 each: Sand=53, Ebonsand=112, Pearlsand=116,
    	// Crimsand=234, plus the Hardened/Sandstone tile families). 09-RESEARCH.md Pitfall 5: Desert's
    	// threshold (1500) is 5-12x higher than every other tile-weighted biome in this project
    	// (Corruption=300, Hallow=125, Jungle=140, Dungeon=250) -- this pass uses thickness=20 (not the
    	// usual 15) for extra safety margin, since the default 15-tile convention clears the threshold
    	// with only ~2x margin instead of Corruption's ~10x.
    	public class DesertPlatformPass : GenPass
    	{
    		public DesertPlatformPass(string name, float loadWeight) : base(name, loadWeight)
    		{
    		}

    		protected override void ApplyPass(GenerationProgress progress, GameConfiguration configuration)
    		{
    			progress.Message = "Generating desert boss arena platform";

    			int surfaceY = Main.maxTilesY / 2; // mid-height, matches CorruptionPlatformPass convention
    			int thickness = 20; // wider margin than the usual 15 -- see class comment (Pitfall 5)

    			for (int x = 0; x < Main.maxTilesX; x++)
    			{
    				for (int y = surfaceY; y < surfaceY + thickness; y++)
    				{
    					Tile tile = Main.tile[x, y];
    					tile.HasTile = true;
    					tile.TileType = TileID.Sand;
    				}
    			}

    			Main.spawnTileX = Main.maxTilesX / 2;
    			Main.spawnTileY = surfaceY - 3;
    		}
    	}
    }
    ```

    ```csharp
    using Terraria;
    using Terraria.ID;
    using Terraria.IO;
    using Terraria.WorldBuilding;

    namespace BossArenaSubWorld.Subworlds
    {
    	// Source: decompiled Terraria.Player.UpdateBiomes() + Terraria.Main.wallDungeon --
    	// ZoneDungeon requires ALL of: SceneMetrics.DungeonTileCount >= SceneMetrics.DungeonTileThreshold
    	// (250, weighted over TileID.Sets.DungeonBiome: BlueDungeonBrick=41/GreenDungeonBrick=43/
    	// PinkDungeonBrick=44), AND Main.wallDungeon[wall at the player's EXACT current tile] == true,
    	// AND player.Center.Y > worldSurface*16. This is the ONLY biome in this project with an extra
    	// per-position WALL requirement layered on top of the weighted tile count (09-RESEARCH.md
    	// Pitfall 3). CRITICAL: only the "Unsafe" wall constants are flagged true in Main.wallDungeon's
    	// initializer (BlueDungeonUnsafe=7, GreenDungeonUnsafe=8, PinkDungeonUnsafe=9, plus 94-99) -- the
    	// player-placeable "safe" Dungeon Brick Wall item is a DIFFERENT WallID and does NOT satisfy this
    	// check, even though it looks visually identical in-game. This pass places WallID.BlueDungeonUnsafe
    	// across the FULL platform width (matching the full-width-tile-fill philosophy already established),
    	// since the player can stand anywhere across the platform.
    	public class DungeonPlatformPass : GenPass
    	{
    		public DungeonPlatformPass(string name, float loadWeight) : base(name, loadWeight)
    		{
    		}

    		protected override void ApplyPass(GenerationProgress progress, GameConfiguration configuration)
    		{
    			progress.Message = "Generating dungeon boss arena platform";

    			int surfaceY = Main.maxTilesY / 2; // 400 at WorldHeight=800 -- > worldSurface (240), satisfies the height half of the check
    			int thickness = 15;

    			for (int x = 0; x < Main.maxTilesX; x++)
    			{
    				for (int y = surfaceY; y < surfaceY + thickness; y++)
    				{
    					Tile tile = Main.tile[x, y];
    					tile.HasTile = true;
    					tile.TileType = TileID.BlueDungeonBrick; // weight 1 in TileID.Sets.DungeonBiome
    					// Wall MUST be an "Unsafe" dungeon wall variant -- the safe/placeable Dungeon
    					// Brick Wall item is NOT in Main.wallDungeon's true set (see class comment).
    					tile.WallType = WallID.BlueDungeonUnsafe; // = 7
    				}
    			}

    			Main.spawnTileX = Main.maxTilesX / 2;
    			Main.spawnTileY = surfaceY - 3;
    		}
    	}
    }
    ```
